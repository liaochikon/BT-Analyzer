# GPU (CUDA) AI / MIDI stack for BT Analyzer.
# Install with:  pip install -r requirements-ai-gpu.txt
# Pulls CUDA builds of torch (+torchaudio/torchvision) and onnxruntime-gpu.
#
# IMPORTANT: the CUDA tag below is cu126 (CUDA 12.6), which has matching wheels for these torch
# versions and works on recent NVIDIA drivers. If your driver is older, pick a tag your driver
# supports that ALSO publishes these torch versions (e.g. cu130 also ships 2.12.0; cu124 only
# goes up to torch 2.6.0). Check:
#   https://pytorch.org/get-started/locally/  and  https://download.pytorch.org/whl/<tag>/torch/
# The +cuXXX local-version pins force the CUDA wheels even though PyPI also ships a CPU 2.12.0.
# MIDI to Audio additionally needs a system FluidSynth install (pretty_midi.fluidsynth).

--extra-index-url https://download.pytorch.org/whl/cu126

torch==2.12.0+cu126            # CUDA 12.6 build
torchaudio==2.11.0+cu126       # matching CUDA build (demucs dependency)
torchvision==0.27.0+cu126      # matching CUDA build
onnxruntime-gpu==1.26.0        # ONNX Runtime w/ CUDAExecutionProvider (do NOT also install onnxruntime)
basic-pitch==0.4.0             # Official Basic Pitch transcription library
ml-dtypes==0.2.0               # TensorFlow 2.15 compatibility on Python 3.11
audio-separator[gpu]==0.30.2   # BS-RoFormer separation on GPU
imageio-ffmpeg==0.5.1          # bundled ffmpeg binary required by audio-separator
demucs==4.0.1                  # Demucs v6 (htdemucs_6s) source separation
pretty_midi==0.2.11            # MIDI to Audio / MIDI Export
